Multilayer Switch

ena
conf t
hostname Multilayer_Switch
spanning-tree mode rapid-pvst

vlan 10
vlan 20
vlan 30
vlan 100
exit

int vlan 1
ip address 192.168.1.1 255.255.255.0
no sh

int vlan 10
ip address 192.168.10.1 255.255.255.0
ip helper-address 192.168.100.150
ip helper-address 192.168.100.149
no sh

int vlan 20
ip address 192.168.20.1 255.255.255.0
ip helper-address 192.168.100.150
ip helper-address 192.168.100.149
no sh

int vlan 30
ip address 192.168.30.1 255.255.255.0
ip helper-address 192.168.100.150
ip helper-address 192.168.100.149
no sh

int vlan 100
ip address 192.168.100.1 255.255.255.0
ip helper-address 192.168.100.150
ip helper-address 192.168.100.149
no sh

ip routing

int gig1/0/3
no switchport
ip address 192.168.2.1 255.255.255.252

int gig1/0/24
no switchport
ip address 192.168.160.1 255.255.255.252

router ospf 1
network 192.168.1.0 0.0.0.255 area 0
network 192.168.10.0 0.0.0.255 area 0
network 192.168.20.0 0.0.0.255 area 0
network 192.168.30.0 0.0.0.255 area 0
network 192.168.100.0 0.0.0.255 area 0
network 192.168.160.0 0.0.0.3 area 0
exit

int range gig1/0/17-18
channel-group 10 mode active
int port-channel 10
switchport mode trunk
switchport trunk allowed vlan 10

int range gig1/0/19-20
channel-group 20 mode active
int port-channel 20
switchport mode trunk
switchport trunk allowed vlan 20

int gig1/0/21
channel-group 30 mode active
int port-channel 30
switchport mode trunk
switchport trunk allowed vlan 30

channel-group 40 mode active
int port-channel 40
switchport mode trunk

exit

switchport access vlan 10
