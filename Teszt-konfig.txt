RH1

ena
conf t
Hostname RH1

int gig0/0/0
ip address 11.11.11.2 255.255.255.252
ip nat outside
no sh

int gig0/0/1
ip address 192.168.1.1 255.255.255.0
ip nat inside
no sh
standby 1 ip 192.168.1.100
standby 1 preempt
standby 1 priority 110

ip route 0.0.0.0 0.0.0.0 gig0/0/0
ip nat inside source static 192.168.150.150 11.11.11.11
ip nat inside source list 1 int gig0/0/0 overload
access-list 1 permit 192.168.1.0 0.0.0.255
access-list 1 permit 192.168.10.0 0.0.0.255
access-list 1 permit 192.168.20.0 0.0.0.255
access-list 1 permit 192.168.40.0 0.0.0.255
access-list 1 permit 192.168.100.0 0.0.0.255
access-list 1 permit 192.168.160.0 0.0.0.255

ip route 192.168.0.0 255.255.0.0 192.168.1.3

router ospf 1
network 192.168.1.0 0.0.0.255 area 0
default-information originate
exit

Multilayer Switch

ena
conf t
hostname Multilayer_Switch
spanning-tree mode rapid-pvst

vlan 2
vlan 10
vlan 20
vlan 30
vlan 100
exit

int vlan 1
ip address 192.168.3.1 255.255.255.0
no sh

int vlan 2
ip address 192.168.1.3 255.255.255.0
no sh

int vlan 10
ip address 192.168.10.1 255.255.255.0
ip helper-address 192.168.100.150
no sh

int vlan 20
ip address 192.168.20.1 255.255.255.0
ip helper-address 192.168.100.150
no sh

int vlan 30
ip address 192.168.30.1 255.255.255.0
ip helper-address 192.168.100.150
no sh

int vlan 100
ip address 192.168.100.1 255.255.255.0
no sh

ip routing

int range gig1/0/1-2
switchport mode access
switchport access vlan 2
exit

int gig1/0/3
no switchport
ip address 192.168.2.1 255.255.255.252

int gig1/0/24
no switchport
ip address 192.168.160.1 255.255.255.252

router ospf 1
network 192.168.1.0 0.0.0.255 area 0
network 192.168.10.0 0.0.0.255 area 0
network 192.168.20.0 0.0.0.255 area 0
network 192.168.30.0 0.0.0.255 area 0
network 192.168.100.0 0.0.0.255 area 0
network 192.168.160.0 0.0.0.3 area 0
exit

int range gig1/0/17-18

int port-channel 1

int range gig1/0/19-20
